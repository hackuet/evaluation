### Task 1: Dual LED Blinker ([`Task-1 LED blinking.ino`](file:///home/user/Projects/HACK%20Evaluation/batches/2k25/assignment-1/submissions/52509016/Task-1%20LED%20blinking.ino))
- **Implementation**:
  ```cpp
  #define LED1_PIN 8 // LED 1 connected to digital pin 8
  #define LED2_PIN 9 // LED 2 connected to digital pin 9

  // Change these values: blink intervals in seconds
  const unsigned long X_SECONDS = 1; // LED 1 toggles every x seconds
  const unsigned long Y_SECONDS = 3; // LED 2 toggles every y seconds

  const unsigned long xInterval = X_SECONDS * 1000UL; // convert to ms
  const unsigned long yInterval = Y_SECONDS * 1000UL;

  unsigned long previousMillis1 = 0;
  unsigned long previousMillis2 = 0;

  int led1State = LOW;
  int led2State = LOW;

  void setup() {
    pinMode(LED1_PIN, OUTPUT);
    pinMode(LED2_PIN, OUTPUT);
  }

  void loop() {
    unsigned long currentMillis = millis();

    // LED 1: toggle every x seconds
    if (currentMillis - previousMillis1 >= xInterval) {
      previousMillis1 = currentMillis;
      led1State = (led1State == LOW) ? HIGH : LOW;
      digitalWrite(LED1_PIN, led1State);
    }

    // LED 2: toggle every y seconds
    if (currentMillis - previousMillis2 >= yInterval) {
      previousMillis2 = currentMillis;
      led2State = (led2State == LOW) ? HIGH : LOW;
      digitalWrite(LED2_PIN, led2State);
    }
  }
  ```
- **Merits**:
  - Clean non-blocking execution using `millis()`.
  - Independent intervals ($1\text{s}$ and $3\text{s}$) executing concurrently.
  - Proper use of `unsigned long` for timestamp comparisons avoiding rollover issues.
- **Score**: **25.00 / 25**

### Task 2: Software PWM ([`Task-2 PWM.ino`](file:///home/user/Projects/HACK%20Evaluation/batches/2k25/assignment-1/submissions/52509016/Task-2%20PWM.ino))
- **Implementation**:
  ```cpp
  #define PWM_PERIOD_US 2000UL
  #define LED_PIN 9

  void pwm(uint8_t pin, uint8_t value) {
    unsigned long pos = micros() % PWM_PERIOD_US;
    unsigned long onTime = (unsigned long)value * PWM_PERIOD_US / 255;
    digitalWrite(pin, (pos < onTime) ? HIGH : LOW);
  }

  uint8_t brightness = 0;
  int step = 1;
  unsigned long lastUpdate = 0;

  void setup() {
    pinMode(LED_PIN, OUTPUT);
  }

  void loop() {
    if (millis() - lastUpdate >= 10) {
      lastUpdate = millis();
      brightness += step;
      if (brightness == 0 || brightness == 255) step = -step;
    }

    pwm(LED_PIN, brightness);
  }
  ```
- **Merits**:
  - Implemented custom `pwm(pin, value)` without `analogWrite()`.
  - High frequency period ($2000\,\mu\text{s} = 500\text{ Hz}$) resulting in flicker-free illumination.
  - Proper cast `(unsigned long)value` prevents 16-bit AVR integer overflow during multiplication ($255 \times 2000 = 510,000$).
  - Built a non-blocking triangle wave breathing animation in `loop()`.
- **Score**: **25.00 / 25**

### Task 3: Sonar Reader ([`Task-3 Sonar reader.ino`](file:///home/user/Projects/HACK%20Evaluation/batches/2k25/assignment-1/submissions/52509016/Task-3%20Sonar%20reader.ino))

  const uint8_t ECHO = 2;

  void setup() {
    Serial.begin(9600);
    pinMode(TRIG, OUTPUT);
    pinMode(ECHO, INPUT);
    digitalWrite(TRIG, LOW);
  }

  void loop() {
    // send trigger pulse
    digitalWrite(TRIG, LOW);
    delayMicroseconds(2);
    digitalWrite(TRIG, HIGH);
    delayMicroseconds(10);
    digitalWrite(TRIG, LOW);

    unsigned long duration = pulseIn(ECHO, HIGH, 30000);

      float cm = duration * 0.0343 / 2.0;
      Serial.print("Distance: ");
      Serial.print(cm);
      Serial.println(" cm");
      digitalWrite(LED_BUILTIN, cm < 20 ? HIGH : LOW);

      Serial.println("No echo received");
    }
    delay(100);
  }
