int ledPin = 9; // LED connected to digital pin 9

// value: Brightness level (0 to 255)
void pwm(int pin, int value) {
  // Setting value within 0 - 255 range
  if (value < 0) value = 0;
  if (value > 255) value = 255;

  // Total time = 10,000 microseconds
  int time = 10000; 

  int onTime = (time * value) / 255; //Duty cycle for onTime 
  int offTime = time - onTime;

  // Execute ON phase
  if (onTime > 0) {
    digitalWrite(pin, HIGH);
    delayMicroseconds(onTime); 
  }

  // Execute OFF phase
  if (offTime > 0) {
    digitalWrite(pin, LOW);
    delayMicroseconds(offTime);
  }
}

void setup() {
  pinMode(ledPin, OUTPUT);
}

void loop() {
  // Gradually increase brightness (Fade In)
  // This condition for increasing brightness with time
  for (int brightness = 0; brightness <= 255; brightness += 5) {
    // Time gap 
    for (int i = 0; i < 10; i++) {
      pwm(ledPin, brightness);
    }
  }

  // Gradually decrease brightness (Fade Out)
  for (int brightness = 255; brightness >= 0; brightness -= 5) {
    for (int i = 0; i < 10; i++) {
      pwm(ledPin, brightness);
    }
  }
}
